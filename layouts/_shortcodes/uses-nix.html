{{- /* /uses as a Nix attribute set, inside the shared terminal frame. Data: data/uses.toml. Values with a url become real links; notes become # comments. */ -}} {{- $d := hugo.Data.uses -}} {{- $o := "" -}} {{- $o = printf `%s
❯ cat uses.nix
` $o -}}
{{- $o = printf `%s# %s` $o (htmlEscape $d.source) -}}
{{- $o = printf "%s\n{\n" $o -}}
{{- range $i, $g := $d.groups -}}
  {{- if $i }}{{ $o = printf "%s\n" $o }}{{ end -}}
  {{- if $g.list -}}
    {{- $o = printf `%s  %s = [` $o $g.name -}}
    {{- range $g.items -}}
      {{- $o = printf "%s\n    " $o -}}
      {{- $name := printf `%s` .url (htmlEscape .key) -}}
      {{- $pad := strings.Repeat (math.Max 1 (sub 10 (len .key))) " " -}}
      {{- $o = printf "%s%s" $o $name -}}
      {{- with .note }}{{ $o = printf `%s%s# %s` $o $pad (htmlEscape .) }}{{ end -}}
    {{- end -}}
    {{- $o = printf "%s\n  ];\n" $o -}}
  {{- else -}}
    {{- $w := 0 }}{{ range $g.items }}{{ $w = math.Max $w (len .key) }}{{ end -}}
    {{- $o = printf `%s  %s = {` $o $g.name -}}
    {{- range $g.items -}}
      {{- $key := printf `%s` (replace .key "." `.`) -}}
      {{- $pad := strings.Repeat (int (sub $w (len .key))) " " -}}
      {{- $val := "" -}}
      {{- if .raw }}{{ $val = printf `%s` (htmlEscape .value) -}}
      {{- else if .url }}{{ $val = printf `"%s"` .url (htmlEscape .value) -}}
      {{- else }}{{ $val = printf `"%s"` (htmlEscape .value) }}{{ end -}}
      {{- $o = printf "%s\n    %s%s = %s;" $o $key $pad $val -}}
      {{- with .note }}{{ $o = printf `%s # %s` $o (htmlEscape .) }}{{ end -}}
    {{- end -}}
    {{- $o = printf "%s\n  };\n" $o -}}
  {{- end -}}
{{- end -}}
{{- $o = printf "%s}
" $o -}} {{- partial "terminal-frame.html" (dict "Content" $o) -}}
