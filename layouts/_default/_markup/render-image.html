{{- $url := urls.Parse .Destination -}} {{- if $url.Scheme -}} {{/* third party image */}} {{- if not (in (slice "http" "https") $url.Scheme) -}} {{- errorf "Unsupported image URL scheme %q in %s" $url.Scheme .Page.File -}} {{- end -}} {{- else -}} {{/* internal image */}} {{- $dest := strings.TrimPrefix "./" .Destination -}} {{- $img := .Page.Resources.GetMatch $dest -}} {{- if not $img -}} {{- $img = resources.Get $dest -}} {{- end -}} {{- if $img -}} {{- if eq $img.MediaType.MainType "video" -}} {{- else -}} {{- end -}} {{- else -}} {{- warnf "No image found for %s from %s" .Destination .Page.File }} {{- end -}} {{- end -}}
