{{- $url := urls.Parse .Destination -}} {{- /* The first image on a page is usually its LCP candidate, so it must not be lazy. */ -}} {{- $loading := cond (eq .Ordinal 0) "eager" "lazy" -}} {{- $priority := cond (eq .Ordinal 0) "high" "auto" -}} {{- if or $url.Scheme $url.Host -}} {{/* third party image */}} {{- if and $url.Scheme (not (in (slice "http" "https") $url.Scheme)) -}} {{- errorf "Unsupported image URL scheme %q in %s" $url.Scheme .Page.File -}} {{- end -}} {{- else -}} {{/* internal image */}} {{- $dest := strings.TrimPrefix "./" $url.Path -}} {{- $img := .Page.Resources.GetMatch $dest -}} {{- if not $img -}} {{- $img = resources.Get (strings.TrimPrefix "/" $dest) -}} {{- end -}} {{- if $img -}} {{- if eq $img.MediaType.MainType "video" -}} {{- $originalGIF := .Page.Resources.GetMatch (printf "%s.gif" (strings.TrimSuffix (path.Ext $dest) $dest)) -}} {{- else -}} {{- $original := $img.RelPermalink -}} {{- $src := $original -}} {{- $sources := slice -}} {{- $processable := reflect.IsImageResourceProcessable $img -}} {{- if and $processable (ne $img.MediaType.SubType "gif") -}} {{- $widths := slice (int (math.Min 720 $img.Width)) (int (math.Min 1440 $img.Width)) | uniq -}} {{- range $i, $width := $widths -}} {{- $resized := $img.Resize (printf "%dx webp q85" $width) -}} {{- if eq $i 0 }}{{ $src = $resized.RelPermalink }}{{ end -}} {{- $sources = $sources | append (printf "%s %dw" $resized.RelPermalink $resized.Width) -}} {{- end -}} {{- end -}} {{- with $url.RawQuery }}{{ $src = printf "%s?%s" $src . }}{{ end -}} {{- with $url.Fragment }}{{ $src = printf "%s#%s" $src . }}{{ end -}} {{- end -}} {{- else -}} {{- warnf "No image found for %s from %s" .Destination .Page.File }} {{- end -}} {{- end -}}
